cmake_minimum_required(VERSION 3.27)
project(AIBenchieBridgeSetup LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
if(NOT DEFINED NULLXOID_DESKTOP_ROOT)
  message(FATAL_ERROR "Set NULLXOID_DESKTOP_ROOT to the desktop source being qualified")
endif()
add_executable(bridge_setup_check setup_check.cpp)
target_include_directories(bridge_setup_check PRIVATE "${NULLXOID_DESKTOP_ROOT}/src")
target_link_libraries(bridge_setup_check PRIVATE Qt6::Core crypt32)
add_executable(credential_sign_in_check credential_sign_in_check.cpp)
target_include_directories(credential_sign_in_check PRIVATE "${NULLXOID_DESKTOP_ROOT}/src")
target_link_libraries(credential_sign_in_check PRIVATE Qt6::Core Qt6::Network)
